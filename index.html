#01: Wireless Server Room Thermometer

// SYSTEM OVERVIEW

This node reads real-time temperature values from a DHT22 sensor and serves them over a localized lightweight web page server via the ESP32 chip module.

Perfect project implementation for monitoring server racks or simple home automation labs.

main_dht22_server.ino 
// BitsToBytes.IOT - Project #01
#include <WiFi.h>

const char* ssid = "BitsToBytes_Net";
const char* password = "SecurePassword123";

void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nNode Online!");
}

void loop() {
  delay(2000);
}
